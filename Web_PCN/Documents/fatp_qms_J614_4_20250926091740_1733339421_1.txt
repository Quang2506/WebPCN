@section Scripts {
    <script>
        // Ripple press visual
        document.querySelectorAll('.btn').forEach(b => {
            b.addEventListener('mousedown', () => b.classList.add('is-pressed'));
            ['mouseup', 'mouseleave', 'blur'].forEach(e => b.addEventListener(e, () => b.classList.remove('is-pressed')));
        });

        // Submit helper
        function submitForApproval() { document.getElementById('crForm').requestSubmit(); }

        // Attachments buffer (add/remove one by one)
        (function () {
            const dz = document.getElementById('dz');
            const input = document.getElementById('fileInput');
            const btnPick = document.getElementById('btnPick');
            const list = document.getElementById('fileList');
            const filesBuf = [];
            const MAX_FILES = 50, MAX_EACH_MB = 50, MAX_TOTAL_MB = 500;

            btnPick.addEventListener('click', () => input.click());
            input.addEventListener('change', () => { addFiles(input.files); input.value = ''; });

            ['dragenter', 'dragover'].forEach(n => dz.addEventListener(n, e => { e.preventDefault(); dz.classList.add('dragover'); }));
            ['dragleave', 'drop'].forEach(n => dz.addEventListener(n, e => { e.preventDefault(); dz.classList.remove('dragover'); }));
            dz.addEventListener('drop', e => addFiles(e.dataTransfer.files));

            function addFiles(files) {
                if (!files || !files.length) return;
                if (filesBuf.length + files.length > MAX_FILES) { alert(`Maximum ${MAX_FILES} files allowed.`); return; }

                let totalBytes = filesBuf.reduce((s, f) => s + f.size, 0);
                for (const f of files) {
                    if (f.size > MAX_EACH_MB * 1024 * 1024) { alert(`"${f.name}" exceeds ${MAX_EACH_MB}MB.`); continue; }
                    if (totalBytes + f.size > MAX_TOTAL_MB * 1024 * 1024) { alert(`Total size exceeds ${MAX_TOTAL_MB}MB.`); continue; }
                    filesBuf.push(f); totalBytes += f.size;
                }
                sync();
            }

            function sync() {
                const dt = new DataTransfer();
                filesBuf.forEach(f => dt.items.add(f));
                input.files = dt.files;

                list.innerHTML = '';
                filesBuf.forEach((f, idx) => {
                    const li = document.createElement('li');
                    li.innerHTML = `<div class="file-info"><strong>${f.name}</strong></div>`;
                    const rm = document.createElement('button');
                    rm.className = 'remove'; rm.type = 'button'; rm.title = 'Remove this file';
                    rm.innerHTML = `
                                <svg width="18" height="18" viewBox="0 0 24 24" fill="none"
                                     xmlns="http://www.w3.org/2000/svg" stroke="currentColor"
                                     stroke-width="2" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true" focusable="false">
                                  <path d="M3 6h18M8 6V4a2 2 0 0 1 2-2h4a2 2 0 0 1 2 2v2"/>
                                  <path d="M19 6l-1 14a2 2 0 0 1-2 2H8a2 2 0 0 1-2-2L5 6"/>
                                  <path d="M10 11v6M14 11v6"/>
                                </svg>`;

                    rm.addEventListener('click', () => { filesBuf.splice(idx, 1); sync(); });
                    li.appendChild(rm); list.appendChild(li);
                });
            }
        })();
    </script>
}
